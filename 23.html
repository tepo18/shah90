n.server, n.port, 2): n for n in self.nodes} for f in futures: n = futures[f] try: n.update(f.result()) except: n.update(None) def compute_active_score(self, node): latency = node.avg_latency() choose_best_proxy(self): # فقط پروکسی‌های پایدار stable_nodes = [n for n in self.nodes if n.stability() >= MIN_STABLE_SCORE and n.alive] if not stable_nodes: return None stable_nodes.sort(key=lambda
